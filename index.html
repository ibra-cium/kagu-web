Types & Variables

                        shongkha x = 10; // int
                        doshomik pi = 3.14; // float
                        lekha name = "Rafi"; // string
                        okkhor c = 'A'; // char
                        
Use lekha for text strings and okkhor for single characters.

                    

                    

                        
Arrays

                        shongkha t sorbochho(5);
                        shongkha t sorbochho(3) = {1, 2, 3};
                        t[0] = 99;
                        dekhao(t[0]);
                    

                    

                        
Pointers & References

                        shongkha pointer p = x.erthikana(); // &x
                        dekhao(p.erman()); // *p
                        p.erman() = 25; // *p = 25
                        shongkha pointer q = faka; // NULL
                    

                    

                        
Operators

                        x += 5; x -= 1; x *= 2; x /= 2; x %= 3;
                        i++; i--;
                        ebong (&&), tasara (||), ulto (!)
                    

                    

                        
Control Flow

                        jodi (a > 0 ebong b > 0) { }
                        nahole jodi (a == 0) { } nahole { }
                        jotokkhon (x < 10) { x++; }
                        chalao (shongkha i = 0; i < 5; i++) { }
                        thamo; // break
                    

                    

                        
Functions & I/O

                        kaj shongkha jog(shongkha a, shongkha b) { ferot a + b; }

                        dekhao("Result: ", res, "\n");

Koto ghor decimal dekhabo

                        doshomik pi = 3.14159;
                        dekhao(pi); // 3.14 (default)
                        dekhao(pi.ghor(0)); // 3
                        dekhao(pi.ghor(2)); // 3.14
                        dekhao(pi.ghor(4)); // 3.1416
                        
Default precision is 2 places. .ghor(n) works on doshomik inside dekhao().
